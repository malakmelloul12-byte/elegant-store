كود Google Apps Script لجدول البيانات

هذا الفورم مربوط مسبقاً بالرابط الخاص بكِ. في حال أردتِ التأكد من أن الأعمدة في Google Sheet تستقبل البيانات بشكل مثالي، ضعي هذا الكود في Extensions > Apps Script: 

function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = e.parameter;
  
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["التاريخ", "الاسم الكامل", "المنتوج", "المقاس", "اللون", "الكمية", "ثمن الوحدة", "المجموع", "المدينة", "العنوان", "رقم الهاتف"]);
  }
  
  sheet.appendRow([
    new Date().toLocaleString("ar-MA"),
    data.fullName || "",
    data.product || "",
    data.size || "",
    data.color || "",
    data.quantity || "",
    data.unitPrice || "",
    data.totalPrice || "",
    data.city || "",
    data.address || "",
    "'" + (data.phone || "")
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({ "result": "success" }))
    .setMimeType(ContentService.MimeType.JSON);
}
